const {user, repo, token} = await getAuth(); const name = document.getElementById('prodName').value; const price = document.getElementById('prodPrice').value; const image = document.getElementById('prodImage').value; const res = await fetch(`https://api.github.com/repos/${user}/${repo}/contents/products.json`, { headers: { "Authorization": `token ${token}` } }); const data = await res.json(); let products = JSON.parse(b64_to_utf8(data.content)); products.push({ id: Date.now(), name, price, image }); await fetch(`https://api.github.com/repos/${user}/${repo}/contents/products.json`, { method: "PUT", headers: { "Authorization": `token ${token}`, "Content-Type": "application/json" }, body: JSON.stringify({ message: "Add product", content: utf8_to_b64(JSON.stringify(products, null, 2)), sha: data.sha }) }); alert("تمت الإضافة!"); loadProducts(); }
